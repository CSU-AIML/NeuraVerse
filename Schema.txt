-- Enhanced user profiles table
CREATE TABLE IF NOT EXISTS user_profiles (
  id BIGSERIAL PRIMARY KEY,
  firebase_uid TEXT UNIQUE NOT NULL,
  email TEXT NOT NULL,
  display_name TEXT,
  avatar_url TEXT,
  role TEXT DEFAULT 'user' CHECK (role IN ('user', 'admin', 'guest')),
  
  -- Account status and security
  account_status TEXT DEFAULT 'active' CHECK (account_status IN ('active', 'suspended', 'deleted', 'pending_verification')),
  email_verified BOOLEAN DEFAULT false,
  two_factor_enabled BOOLEAN DEFAULT false,
  password_changed_at TIMESTAMP WITH TIME ZONE,
  last_login TIMESTAMP WITH TIME ZONE,
  last_ip INET,
  login_count INTEGER DEFAULT 0,
  failed_login_attempts INTEGER DEFAULT 0,
  account_locked_until TIMESTAMP WITH TIME ZONE,
  
  -- Suspension management
  suspension_reason TEXT,
  suspension_end_date TIMESTAMP WITH TIME ZONE,
  suspended_by BIGINT REFERENCES user_profiles(id),
  
  -- GDPR and data compliance

  gdpr_consent_date TIMESTAMP WITH TIME ZONE,
  data_retention_date TIMESTAMP WITH TIME ZONE,
  data_processing_consent BOOLEAN DEFAULT false,
  marketing_consent BOOLEAN DEFAULT false,
  
  -- Audit fields
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
  created_by BIGINT REFERENCES user_profiles(id),
  updated_by BIGINT REFERENCES user_profiles(id),
  
  -- Soft delete support
  deleted_at TIMESTAMP WITH TIME ZONE,
  deleted_by BIGINT REFERENCES user_profiles(id),
  deletion_reason TEXT
);

-- Audit logs table for compliance and security monitoring
CREATE TABLE IF NOT EXISTS audit_logs (
  id UUID DEFAULT uuid_generate_v4() PRIMARY KEY,
  action TEXT NOT NULL,
  actor_id TEXT NOT NULL, -- Firebase UID of person performing action
  target_user_id TEXT, -- Firebase UID of user being affected
  target_resource_type TEXT, -- 'user', 'project', 'system', etc.
  target_resource_id TEXT,
  
  -- Request context
  ip_address INET,
  user_agent TEXT,
  session_id TEXT,
  request_id TEXT,
  
  -- Action details
  details JSONB DEFAULT '{}',
  severity TEXT DEFAULT 'medium' CHECK (severity IN ('low', 'medium', 'high', 'critical')),
  
  -- Success/failure tracking
  success BOOLEAN DEFAULT true,
  error_message TEXT,
  
  -- Compliance
  retention_date TIMESTAMP WITH TIME ZONE DEFAULT (NOW() + INTERVAL '7 years'),
  
  created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS user_sessions (

  session_token TEXT UNIQUE NOT NULL,
  refresh_token TEXT,
  
  -- Session metadata

  user_agent TEXT,
  device_fingerprint TEXT,

-- Stored procedures for common operations
CREATE OR REPLACE FUNCTION check_user_permission(
  user_firebase_uid TEXT,
  permission_name TEXT
)
RETURNS BOOLEAN AS $$
DECLARE
  user_role TEXT;
  has_permission BOOLEAN := FALSE;
BEGIN
  -- Get user role
  SELECT role INTO user_role
  FROM user_profiles
  WHERE firebase_uid = user_firebase_uid
  AND account_status = 'active';
  
  -- Admins have all permissions
  IF user_role = 'admin' THEN
    RETURN TRUE;
  END IF;
  
  -- Check specific permission
  SELECT EXISTS (
    SELECT 1
    FROM user_permissions up
    JOIN permissions p ON up.permission_id = p.id
    JOIN user_profiles u ON up.user_id = u.id
    WHERE u.firebase_uid = user_firebase_uid
    AND p.name = permission_name
    AND (up.expires_at IS NULL OR up.expires_at > NOW())
  ) INTO has_permission;
  
  RETURN has_permission;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function to log user activity
CREATE OR REPLACE FUNCTION log_user_activity()
RETURNS TRIGGER AS $$
BEGIN
  IF TG_OP = 'UPDATE' THEN
    -- Log significant profile changes
    IF OLD.role != NEW.role OR OLD.account_status != NEW.account_status THEN
      INSERT INTO audit_logs (
        action,
        actor_id,
        target_user_id,
        details
      ) VALUES (
        'user_profile_updated',
        auth.uid(),
        NEW.firebase_uid,
        jsonb_build_object(
          'old_role', OLD.role,
          'new_role', NEW.role,
          'old_status', OLD.account_status,
          'new_status', NEW.account_status,
          'updated_fields', array(
            SELECT key FROM jsonb_each_text(to_jsonb(NEW)) 
            WHERE to_jsonb(NEW) -> key != to_jsonb(OLD) -> key
          )
        )
      );
    END IF;
  END IF;
  
  RETURN COALESCE(NEW, OLD);
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Trigger for user activity logging
DROP TRIGGER IF EXISTS trigger_log_user_activity ON user_profiles;
CREATE TRIGGER trigger_log_user_activity
  AFTER UPDATE ON user_profiles
  FOR EACH ROW
  EXECUTE FUNCTION log_user_activity();

-- Function to clean up expired sessions
CREATE OR REPLACE FUNCTION cleanup_expired_sessions()
RETURNS INTEGER AS $$
DECLARE
  deleted_count INTEGER;
BEGIN
  DELETE FROM user_sessions 
  WHERE expires_at < NOW() 
  OR revoked_at IS NOT NULL;
  
  GET DIAGNOSTICS deleted_count = ROW_COUNT;
  RETURN deleted_count;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Function for GDPR data export
CREATE OR REPLACE FUNCTION export_user_data(user_firebase_uid TEXT)
RETURNS JSONB AS $$
DECLARE
  user_data JSONB;
  profile_data JSONB;
  preferences_data JSONB;
  audit_data JSONB;
BEGIN
  -- Get user profile
  SELECT to_jsonb(up.*) INTO profile_data
  FROM user_profiles up
  WHERE up.firebase_uid = user_firebase_uid;
  
  -- Get user preferences
  SELECT to_jsonb(pref.*) INTO preferences_data
  FROM user_preferences pref
  JOIN user_profiles up ON pref.user_id = up.id
  WHERE up.firebase_uid = user_firebase_uid;
  
  -- Get recent audit logs (last 100 entries)
  SELECT jsonb_agg(al.*) INTO audit_data
  FROM (
    SELECT * FROM audit_logs 
    WHERE target_user_id = user_firebase_uid 
    ORDER BY created_at DESC 
    LIMIT 100
  ) al;
  
  -- Combine all data
  user_data := jsonb_build_object(
    'export_date', NOW(),
    'user_profile', COALESCE(profile_data, '{}'::jsonb),
    'preferences', COALESCE(preferences_data, '{}'::jsonb),
    'audit_history', COALESCE(audit_data, '[]'::jsonb),
    'data_controller', 'NeuraVerse Platform',
    'export_format_version', '1.0'
  );
  
  RETURN user_data;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

    JOIN permissions p ON up.permission_id = p.id
    JOIN user_profiles u ON up.user_id = u.id
    WHERE u.firebase_uid = user_firebase_uid
    AND p.name = permission_name
    AND (up.expires_at IS NULL OR up.expires_at > NOW())
  ) INTO has_permission;
  
  RETURN has_permission;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

-- Create a function to get the best image URL for a project
CREATE OR REPLACE FUNCTION get_project_image_url(
    p_image_path TEXT,
    p_image_url TEXT,
    p_supabase_url TEXT DEFAULT 'https://your-project.supabase.co'
)
RETURNS TEXT
LANGUAGE plpgsql
AS $$
BEGIN
    -- Priority 1: Supabase Storage path
    IF p_image_path IS NOT NULL AND p_image_path != '' THEN
        RETURN p_supabase_url || '/storage/v1/object/public/project-images/' || p_image_path;
    END IF;
    
    -- Priority 2: External URL
    IF p_image_url IS NOT NULL AND p_image_url != '' THEN
        RETURN p_image_url;
    END IF;
    
    -- No image available
    RETURN NULL;
END;
$$;

-- Create a view for projects with computed image URLs
CREATE OR REPLACE VIEW projects_with_images AS
SELECT 
    p.*,
    get_project_image_url(p.image_path, p.image_url) as computed_image_url,
    CASE 
        WHEN p.image_path IS NOT NULL THEN 'storage'
        WHEN p.image_url IS NOT NULL THEN 'external'
        ELSE 'none'
    END as image_source_type,
    CASE 
        WHEN p.image_path IS NOT NULL OR p.image_url IS NOT NULL THEN true
        ELSE false
    END as has_image
FROM projects p;

-- Update the cleanup trigger to handle both image types
CREATE OR REPLACE FUNCTION cleanup_project_images()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    -- Clean up Supabase Storage images
    IF OLD.image_path IS NOT NULL AND OLD.image_path != '' THEN
        -- Delete the image file from storage if it exists
        DELETE FROM storage.objects 
        WHERE bucket_id = 'project-images' 
        AND name = OLD.image_path;
        
        -- Log the cleanup (optional)
        INSERT INTO project_image_cleanup_log (project_id, image_path, cleanup_date)
        VALUES (OLD.id, OLD.image_path, NOW())
        ON CONFLICT DO NOTHING;
    END IF;
    
    RETURN OLD;
END;
$$;

-- Create cleanup log table (optional)
CREATE TABLE IF NOT EXISTS project_image_cleanup_log (
    id SERIAL PRIMARY KEY,
    project_id TEXT NOT NULL,
    image_path TEXT,
    cleanup_date TIMESTAMP DEFAULT NOW()
);

-- Update the trigger
DROP TRIGGER IF EXISTS cleanup_project_image_trigger ON projects;
CREATE TRIGGER cleanup_project_images_trigger
    AFTER DELETE ON projects
    FOR EACH ROW
    EXECUTE FUNCTION cleanup_project_images();

-- Create a function to migrate external URLs to Supabase Storage (optional)
CREATE OR REPLACE FUNCTION migrate_project_images_to_storage()
RETURNS TABLE(project_id TEXT, old_url TEXT, migration_status TEXT)
LANGUAGE plpgsql
AS $$
BEGIN
    RETURN QUERY
    SELECT 
        p.id,
        p.image_url,
        CASE 
            WHEN p.image_url IS NOT NULL AND p.image_path IS NULL THEN 'needs_migration'
            WHEN p.image_url IS NOT NULL AND p.image_path IS NOT NULL THEN 'migrated'
            WHEN p.image_path IS NOT NULL THEN 'storage_only'
            ELSE 'no_image'
        END
    FROM projects p
    ORDER BY 
        CASE 
            WHEN p.image_url IS NOT NULL AND p.image_path IS NULL THEN 1
            ELSE 2
        END;
END;
$$;
